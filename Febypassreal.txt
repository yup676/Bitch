local function esc(s)
	s = tostring(s or "")
	return (s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;"))
end

local function sanitize(s)
	if type(s) ~= "string" then return tostring(s or "") end
	s = s:gsub("[%c]", function(c)
		if c == "\n" or c == "\t" then return c end
		return ""
	end)
	return s:gsub("\r\n", "\n"):gsub("\r", "\n")
end

local function truncate(s, m)
	s = tostring(s or "")
	m = m or 6000
	if #s <= m then return s end
	return s:sub(1, m) .. "\n[...+" .. (#s - m) .. " chars]"
end

local function safeGet(obj, prop, default)
	if not obj then return default end
	local ok, v = pcall(function() return obj[prop] end)
	if ok then return v end
	return default
end

local function safeClassName(obj)
	local n = safeGet(obj, "ClassName")
	return type(n) == "string" and n or "<unknown>"
end

local function safeFullName(obj)
	local ok, n = pcall(function() return obj:GetFullName() end)
	if not ok or type(n) ~= "string" then return "<no-path>" end
	return (n:gsub("[^\32-\126]", "?"):gsub("?+", "?"))
end

local function safeEncode(tbl)
	local ok, res = pcall(function() return httpService:JSONEncode(tbl) end)
	if ok and type(res) == "string" and #res > 0 then return res end
	return "{}"
end

local function httpRequest(req, timeout)
	timeout = timeout or HTTP_TIMEOUT
	local done, result, err = false, nil, nil
	task.spawn(function()
		local ok, res = pcall(request, req)
		if ok then result = res else err = tostring(res) end
		done = true
	end)
	local start = tick()
	while not done and (tick() - start) < timeout do task.wait(0.05) end
	if not done then return nil, "timeout " .. timeout .. "s" end
	if err then return nil, err end
	return result, nil
end

local function waitForSlot()
	local elapsed = tick() - lastRequestAt
	if elapsed < MIN_REQUEST_INTERVAL then
		task.wait(MIN_REQUEST_INTERVAL - elapsed)
	end
	lastRequestAt = tick()
end

local function maybeWaitForRateLimit()
	local now = tick()
	if rateLimitUntil > now then
		local w = rateLimitUntil - now
		if w > 0.1 then
			appendOutput("\n<font color=\"#666677\">[log] rate limit " .. string.format("%.1f", w) .. "s</font>")
			task.wait(w)
		end
	end
end

local function scanRemotesAndScripts(onProgress)
	local ev, fn, bd = {}, {}, {}
	local locals, servers, modules = {}, {}, {}
	local evTotal, fnTotal, bdTotal = 0, 0, 0
	local locTotal, srvTotal, modTotal = 0, 0, 0

	local roots = {
		replicatedStorage, workspaceService, replicatedFirst, starterGui,
		starterPack, starterPlayer, lighting,
	}
	pcall(function() table.insert(roots, game:GetService("ServerStorage")) end)
	pcall(function() table.insert(roots, game:GetService("ServerScriptService")) end)
	pcall(function() table.insert(roots, game:GetService("Chat")) end)
	pcall(function() table.insert(roots, game:GetService("SoundService")) end)
	pcall(function() table.insert(roots, game:GetService("Teams")) end)
	pcall(function() table.insert(roots, game:GetService("TextChatService")) end)

	local processed = 0
	for _, root in ipairs(roots) do
		local ok, descendants = pcall(function() return root:GetDescendants() end)
		if ok and descendants then
			local batch = 0
			for i = 1, #descendants do
				local d = descendants[i]
				local cls = safeClassName(d)
				if cls == "RemoteEvent" or cls == "UnreliableRemoteEvent" then
					evTotal = evTotal + 1
					if #ev < 80 then table.insert(ev, d) end
				elseif cls == "RemoteFunction" then
					fnTotal = fnTotal + 1
					if #fn < 80 then table.insert(fn, d) end
				elseif cls == "BindableEvent" or cls == "BindableFunction" then
					bdTotal = bdTotal + 1
					if #bd < 40 then table.insert(bd, d) end
				elseif cls == "LocalScript" then
					locTotal = locTotal + 1
					if #locals < 50 then table.insert(locals, d) end
				elseif cls == "Script" then
					srvTotal = srvTotal + 1
					if #servers < 50 then table.insert(servers, d) end
				elseif cls == "ModuleScript" then
					modTotal = modTotal + 1
					if #modules < 50 then table.insert(modules, d) end
				end
				processed = processed + 1
				if processed % SCAN_BATCH == 0 then
					batch = batch + 1
					if batch >= SCAN_YIELD_EVERY then
						batch = 0
						if onProgress then onProgress(processed, #descendants) end
						task.wait()
						if shuttingDown then return end
					end
				end
			end
		end
	end

	gameScan.remotes.events = ev
	gameScan.remotes.functions = fn
	gameScan.remotes.bindables = bd
	gameScan.remotes.total = evTotal + fnTotal + bdTotal
	gameScan.scripts.locals = locals
	gameScan.scripts.server = servers
	gameScan.scripts.modules = modules
	gameScan.scripts.total = locTotal + srvTotal + modTotal
end

local function scanWorkspace()
	local parts, models, tools = 0, 0, 0
	local top, npcs = {}, {}
	local ok, children = pcall(function() return workspaceService:GetChildren() end)
	if ok and children then
		for i = 1, math.min(#children, 12) do
			local c = children[i]
			top[#top + 1] = c.Name:gsub("[^\32-\126]", "?") .. " (" .. safeClassName(c) .. ")"
		end
	end
	local ok2, desc = pcall(function() return workspaceService:GetDescendants() end)
	if ok2 and desc then
		local batch = 0
		for i = 1, #desc do
			local cls = safeClassName(desc[i])
			if cls == "Part" or cls == "MeshPart" or cls == "UnionOperation" or cls == "WedgePart" or cls == "CornerWedgePart" or cls == "TrussPart" or cls == "SpawnLocation" then
				parts = parts + 1
			elseif cls == "Model" then
				models = models + 1
				local name = safeGet(desc[i], "Name") or ""
				if name:lower():find("npc") or name:lower():find("dummy") or name:lower():find("enemy") then
					if #npcs < 15 then npcs[#npcs + 1] = safeFullName(desc[i]) end
				end
			elseif cls == "Tool" then
				tools = tools + 1
			end
			if i % SCAN_BATCH == 0 then
				batch = batch + 1
				if batch >= SCAN_YIELD_EVERY then
					batch = 0
					task.wait()
					if shuttingDown then return end
				end
			end
		end
	end
	gameScan.workspace.parts = parts
	gameScan.workspace.models = models
	gameScan.workspace.tools = tools
	gameScan.workspace.top = top
	gameScan.workspace.npcs = npcs
end

local function scanPlayers()
	local list = {}
	for _, p in ipairs(playersService:GetPlayers()) do
		local entry = { name = p.Name, userId = p.UserId, display = p.DisplayName }
		local char = p.Character
		if char then
			local hum = char:FindFirstChildOfClass("Humanoid")
			if hum then
				entry.health = hum.Health
				entry.maxHealth = hum.MaxHealth
				entry.walkSpeed = hum.WalkSpeed
				entry.jumpPower = hum.JumpPower
			end
			local hrp = char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local pos = hrp.Position
				entry.position = string.format("%.0f,%.0f,%.0f", pos.X, pos.Y, pos.Z)
			end
		end
		list[#list + 1] = entry
	end
	gameScan.players.list = list
	gameScan.players.total = #list
end

local function scanContainers()
	local result = {}
	local names = {
		"ReplicatedStorage", "ReplicatedFirst", "ServerStorage", "ServerScriptService",
		"StarterGui", "StarterPack", "StarterPlayer", "Workspace", "Lighting",
		"SoundService", "Chat", "TextChatService", "Teams", "Players",
	}
	for _, name in ipairs(names) do
		local ok, svc = pcall(function() return game:GetService(name) end)
		if ok and svc then
			local ok2, ch = pcall(function() return svc:GetChildren() end)
			if ok2 and ch then
				local entry = { name = name, childCount = #ch, children = {} }
				for i = 1, math.min(#ch, 10) do
					entry.children[#entry.children + 1] = ch[i].Name:gsub("[^\32-\126]", "?") .. "(" .. safeClassName(ch[i]) .. ")"
				end
				result[name] = entry
			end
		end
	end
	gameScan.containers = result
end

local pathCache = {}
local function cachePath(p, inst) pathCache[p] = { inst = inst, t = tick() } end
local function getCachedPath(p)
	local e = pathCache[p]
	if not e then return nil end
	if (tick() - e.t) > 6 then pathCache[p] = nil return nil end
	local ok = pcall(function() return e.inst.Parent end)
	if not ok then pathCache[p] = nil return nil end
	return e.inst
end

local function findInstanceByPath(path)
	path = tostring(path or ""):gsub("^%s+", ""):gsub("%s+$", "")
	local cached = getCachedPath(path)
	if cached then return cached end
	local original = path
	if path:sub(1, 5) == "game." then path = path:sub(6) end
	if path == "game" then cachePath(original, game) return game end
	local parts = {}
	for p in path:gmatch("[^%.]+") do parts[#parts + 1] = p end
	if #parts == 0 then return nil end
	local root
	local ok, svc = pcall(function() return game:GetService(parts[1]) end)
	if ok and svc then root = svc else root = game:FindFirstChild(parts[1]) end
	if not root then return nil end
	for i = 2, #parts do
		local nx = root:FindFirstChild(parts[i])
		if not nx then return nil end
		root = nx
	end
	cachePath(original, root)
	return root
end

local function executeAction(input)
	input = tostring(input or ""):gsub("^%s+", ""):gsub("%s+$", "")
	local parts = {}
	for p in input:gmatch("%S+") do parts[#parts + 1] = p end
	if #parts == 0 then return "sem comando" end
	local cmd = parts[1]:lower()
	local fn = actions[cmd]
	if not fn then
		local names = {}
		for k in pairs(actions) do table.insert(names, k) end
		table.sort(names)
		return "comando '" .. cmd .. "' invalido. use: " .. table.concat(names, ", ")
	end
	local args = {}
	for i = 2, #parts do table.insert(args, parts[i]) end
	local ok, result = pcall(fn, table.unpack(args))
	if not ok then return "erro: " .. tostring(result) end
	return truncate(tostring(result))
end

local function fetchModelList()
	local res, err = httpRequest({
		Url = MODELS_URL,
		Method = "GET",
		Headers = { ["Authorization"] = "Bearer " .. API_KEY },
	}, 5)
	if not res then return nil, "req: " .. tostring(err) end
	if res.StatusCode ~= 200 then return nil, "HTTP " .. tostring(res.StatusCode) end
	local ok, dec = pcall(function() return httpService:JSONDecode(res.Body) end)
	if not ok or not dec or not dec.data then return nil, "invalida" end
	local list = {}
	for _, m in ipairs(dec.data) do
		if m.id then table.insert(list, m.id) end
	end
	return list, nil
end

local function pickModels()
	local list, err = fetchModelList()
	if list then
		local available = {}
		for _, id in ipairs(list) do available[id] = true end
		local filtered = {}
		for _, id in ipairs(MODEL_PREFER) do
			if available[id] then table.insert(filtered, id) end
		end
		if #filtered > 0 then
			log("modelos: " .. #filtered)
			return filtered
		end
		return list
	end
	log("fallback cego (" .. tostring(err) .. ")")
	return MODEL_PREFER
end

-- IA CALL
-- ============================================================
local function tryModels(messages)
	local lastErr = nil
	local saw429 = false
	lastRetryAfter = nil
	for _, model in ipairs(verifiedModels) do
		if shuttingDown then return nil, "shutdown" end
		maybeWaitForRateLimit()
		waitForSlot()
		local body = safeEncode({
			model = model,
			messages = messages,
			temperature = TEMPERATURE,
			max_tokens = MAX_TOKENS,
		})
		local resp, terr = httpRequest({
			Url = API_URL,
			Method = "POST",
			Headers = {
				["Content-Type"] = "application/json",
				["Authorization"] = "Bearer " .. API_KEY,
			},
			Body = body,
		}, HTTP_TIMEOUT)
		if not resp then
			lastErr = "req: " .. tostring(terr)
		elseif resp.StatusCode == 200 then
			local ok, dec = pcall(function() return httpService:JSONDecode(resp.Body) end)
			if ok and dec and dec.choices and dec.choices[1] then
				local content = (dec.choices[1].message and dec.choices[1].message.content) or ""
				if #content:gsub("%s", "") > 0 then
					if activeModel ~= model then activeModel = model log("modelo -> " .. model) end
					rateLimitUntil = 0
					return content, nil
				end
				lastErr = "vazio"
			else
				lastErr = "parse"
			end
		elseif resp.StatusCode == 429 then
			saw429 = true
			lastErr = "429"
			if resp.Headers then
				local ra = resp.Headers["retry-after"] or resp.Headers["Retry-After"]
				if ra then
					lastRetryAfter = tonumber(ra)
					if lastRetryAfter then rateLimitUntil = tick() + lastRetryAfter end
				end
			end
		elseif resp.StatusCode == 401 then
			return nil, "invalid-key"
		elseif resp.StatusCode == 400 then
			lastErr = "400: " .. tostring(resp.Body):sub(1, 200)
		else
			lastErr = "HTTP " .. tostring(resp.StatusCode)
		end
	end
	if saw429 then return nil, "RATE_LIMIT" end
	return nil, lastErr or "falha"
end

local function askIA(messages)
	local safe = {}
	for i, m in ipairs(messages) do
		safe[i] = { role = sanitize(m.role), content = sanitize(m.content) }
	end

	for attempt = 1, MAX_RETRY_ATTEMPTS do
		local answer, err = tryModels(safe)
		if not answer then
			if err ~= "RATE_LIMIT" then return nil, err end
			local w = lastRetryAfter or ((2 ^ (attempt - 1)) * BASE_BACKOFF + math.random(0, 2))
			appendOutput("\n<font color=\"#666677\">[log] 429, " .. math.ceil(w) .. "s</font>")
			task.wait(w)
		else
			return answer, nil
		end
	end
	return nil, "RATE_LIMIT_FINAL"
end

-- AGENT LOOP
-- ============================================================
local memory = { history = {} }

local function trimMemory()
	while #memory.history > MAX_HISTORY do table.remove(memory.history, 1) end
end

local function runAgentLoop(question)
	local messages = { { role = "system", content = SYSTEM_PROMPT } }
	for _, m in ipairs(memory.history) do
		if type(m) == "table" and m.role and m.content then
			table.insert(messages, { role = m.role, content = m.content })
		end
	end

	table.insert(messages, {
		role = "user",
		content = "CONTEXTO DO JOGO:\n" .. contextSummary() .. "\n\n(se precisar, use [EXEC])\n\nPERGUNTA: " .. question,
	})

	local maxSteps = 3
	for step = 1, maxSteps do
		local answer, err = askIA(messages)
		if not answer then return nil, err end

		local execCmd = answer:match("^%s*%[EXEC%]%s*(.-)%s*$")
		if not execCmd then
			table.insert(memory.history, { role = "user", content = question })
			table.insert(memory.history, { role = "assistant", content = answer })
			trimMemory()
			return answer, nil
		end

		uiLog("exec: " .. execCmd)
		local ok, result = pcall(executeAction, execCmd)
		if not ok then result = "erro: " .. tostring(result) end

		table.insert(messages, { role = "assistant", content = answer })
		table.insert(messages, {
			role = "user",
			content = "RESULTADO:\n" .. result .. "\n\nAgora responde com base nisso.",
		})
	end
	return nil, "limite de execucoes"
end

local busy = false

local function onSend()
	if busy or shuttingDown then return end
	local q = (input.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
	if #q == 0 then return end

	busy = true
	input.Text = ""
	btnSend.Text = "..."
	uiUser(q)

	local lower = q:lower()

	-- comando direto: "acao: <cmd>" ou "!<cmd>"
	if lower:sub(1, 6) == "acao: " or lower:sub(1, 6) == "acao:" then
		local cmd = q:sub(6):gsub("^%s+", "")
		task.spawn(function()
			local r = executeAction(cmd)
			appendOutput("\n<font color=\"#55cc55\">$ " .. esc(cmd) .. "</font>\n" .. esc(r))
			btnSend.Text = "SEND"
			busy = false
		end)
		return
	end

	if q:sub(1, 1) == "!" then
		local cmd = q:sub(2)
		task.spawn(function()
			local r = executeAction(cmd)
			appendOutput("\n<font color=\"#55cc55\">$ " .. esc(cmd) .. "</font>\n" .. esc(r))
			btnSend.Text = "SEND"
			busy = false
		end)
		return
	end

	task.spawn(function()
		if not gameScan.scanned then
			uiLog("scan rapido...")
			pcall(runFullScan)
		end

		local t0 = tick()
		local answer, err = runAgentLoop(q)
		if answer then
			uiAnswer(answer)
			uiLog(string.format("%.2fs | %s", tick() - t0, activeModel or "?"))
		elseif err == "RATE_LIMIT_FINAL" then
			appendOutput("\n<font color=\"#ff5555\">[rate limit - aguarde 30s]</font>")
		else
			appendOutput("\n<font color=\"#ff5555\">[erro: " .. esc(tostring(err)) .. "]</font>")
		end
		btnSend.Text = "SEND"
		busy = false
	end)
end

task.spawn(function()
	setProgress(0.05, "iniciando...", "")

	setProgress(0.10, "conectando IA...", "")
	local models = pickModels()
	if not models or #models == 0 then
		setProgress(1, "ERRO", "sem modelos")
		task.wait(2)
		closeLoading()
		return
	end
	verifiedModels = models
	activeModel = verifiedModels[1]

	setProgress(0.20, "escaneando DataModel...", "")
	setStats("remotes / scripts / workspace / players / guis")

	runFullScan()

	setProgress(1, "pronto", string.format("%.2fs | %s", gameScan.scanTime, activeModel))
	task.wait(0.3)
	screenGui.Enabled = true
	closeLoading()

	appendOutput("<font color=\"#8a8a95\">Fedora Private IA pronta.</font>")
	appendOutput("\n<font color=\"#8a8a95\">" .. esc(contextSummary()) .. "</font>")
	appendOutput("\n\nDigite uma pergunta, ou use:")
	appendOutput("\n<font color=\"#8a8a95\">  !remotes   !scripts   !guis   !players   !find <nome>   !list <path>   !info <path></font>")
	appendOutput("\n<font color=\"#8a8a95\">  !fire <idx> <args>   !invoke <idx> <args>   !contexto_completo</font>")
	updateCanvas()

	task.spawn(function()
		while not shuttingDown do
			task.wait(REMOTE_TTL)
			if shuttingDown then break end
			pcall(function() scanRemotesAndScripts(nil) end)
		end
	end)
end)

log("boot private scanner pronto")
